 * [92] 反转链表 II

#include "leetcode.h"
// @lc code=start
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseBetween(ListNode* head, int left, int right) {
        if(head == nullptr || left == right) 
        {
            return head;
        }

        ListNode dump(0);
        dump.next = head;

        // pre_node为反转 left 位置的前一个
        ListNode* pre_node = &dump;
        for(int i = 1; i < left  && pre_node != nullptr; i++) 
        {
            pre_node = pre_node ->next;
        }
        // 说明链表没这么长
        if (pre_node == nullptr || pre_node->next == nullptr)
        {
            return head;
        }

        // 反转后这个就是尾部的node
        ListNode* tail_node = pre_node->next;

        // 现在开始找righ位置后的下一个的node
        ListNode* right_pre_node = tail_node;
        for(int i = left; i < right  && right_pre_node  != nullptr; i++) 
        {
            right_pre_node = right_pre_node ->next;
        }
        // 说明rignt没这么长
        if (right_pre_node == nullptr)
        {
            return head;
        }
        ListNode* right_node = right_pre_node->next;
        right_pre_node->next = nullptr;

        // 反转
        ListNode* reverse_list_head = reverseList(tail_node);
        pre_node->next = reverse_list_head;
        tail_node->next = right_node;
        return dump.next;
    }

    ListNode* reverseList(ListNode* head) 
    {
        ListNode* pre = nullptr;
        while(head != nullptr)
        {
            ListNode* next = head->next;
            head->next = pre;
            pre = head;
            head = next;
        }
        return pre;
    }
};
// @lc code=end
