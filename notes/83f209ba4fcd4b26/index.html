23.合并-k-个升序链表.cpp
/*
 * @lc app=leetcode.cn id=23 lang=cpp
 *
 * [23] 合并 K 个升序链表
 */
#include "leetcode.h"
// @lc code=start
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* merge2List(ListNode* l1, ListNode* l2) {
        if (!l1)
        {
            return l2;
        }
        if (!l2)
        {
            return l1;
        }
        ListNode*& cur = (l1->val > l2->val) ? l2 : l1;
        ListNode* head = cur;
        cur = cur->next;
        head->next = merge2List(l1, l2);
        return head;
    }

    ListNode* mergeKList(vector<ListNode*>& lists, int start, int end) {
        if (end < start)
        {
            return nullptr;
        }
        if (end == start)
        {
            return lists[start];
        }
       int mid = (start + end) / 2;
       ListNode* left = mergeKList(lists, start, mid);
       ListNode* right = mergeKList(lists, mid + 1, end);
       return merge2List(left, right);
    }

    ListNode* mergeKLists(vector<ListNode*>& lists) {
        return mergeKList(lists, 0, lists.size() - 1);;
    }
    
};
// @lc code=end
