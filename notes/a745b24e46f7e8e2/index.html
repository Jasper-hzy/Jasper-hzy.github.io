143.重排链表.cpp
/*
 * @lc app=leetcode.cn id=143 lang=cpp
 *
 * [143] 重排链表
 */
#include "leetcode.h"
// @lc code=start
class Solution {
public:
    void reorderList(ListNode* head) {
        if(head == nullptr || head->next == nullptr)
        {
            return;
        }

        ListNode* fast = head;
        ListNode* slow = head;
        // 找到中间节点
        while(fast->next != nullptr && fast->next->next != nullptr)
        {
            slow = slow->next;
            fast = fast->next->next;
        }

        // 拆分后半部分，反转它
        ListNode* half_node = slow->next;
        slow->next = nullptr;
        ListNode* previous = revertNode(half_node);

        ListNode* first = head;
        ListNode* second = previous;
        // 交替拆分
        while(second != nullptr)
        {
            ListNode* firstNext = first->next;
            ListNode* secondNext = second->next;

            first->next = second;
            second->next = firstNext;

            first = firstNext;
            second = secondNext;
        }
    }

    ListNode* revertNode(ListNode* head) {
        if(head == nullptr || head->next == nullptr) 
        {
            return head;
        }
        ListNode* child_node = revertNode(head->next);
        head->next->next = head;
        head->next = nullptr;
        return child_node;
    }
};
// @lc code=end
