33.搜索旋转排序数组

33.搜索旋转排序数组.cpp
/*
 * @lc app=leetcode.cn id=33 lang=cpp
 *
 * [33] 搜索旋转排序数组
 */

#include "leetcode.h"

// @lc code=start
class Solution {
public:
    int searchDps(vector<int>& nums, int target, int l, int r)
    {
        if(l > r) {
            return -1;
        }
        int mid = l + (r - l) / 2;
        if (nums[mid] == target) 
        {
            return mid;
        }

        if(nums[mid] >= nums[l])
        {   // 左边有序
            if(nums[mid] > target && target >= nums[l])
            {
                return searchDps(nums, target, l, mid - 1);
            }
            else 
            {
                return searchDps(nums, target, mid + 1, r);
            }

        } else
        {   // 右边有序
            if(nums[mid] < target && target <= nums[r])
            {
                return searchDps(nums, target, mid + 1, r);
            }
            else 
            {
                return searchDps(nums, target, l, mid - 1);
            }
        }
    }

    int search(vector<int>& nums, int target)
    {
       return searchDps(nums, target, 0, nums.size() - 1); 
    }


};
// @lc code=end
