387.字符串中的第一个唯一字符

保留题目对应的 C++ 练习实现，便于回顾与继续练习。

387.字符串中的第一个唯一字符.cpp
/*
 * @lc app=leetcode.cn id=387 lang=cpp
 *
 * [387] 字符串中的第一个唯一字符
 */
#include "leetcode.h"
// @lc code=start
class Solution
{
public:
    int firstUniqChar(string s)
    {
        int count[26] = {};
        for (int i = 0; i < s.size(); i++)
        {
            count[s[i] - 'a']++;
        }

        for (int i = 0; i < s.size(); i++)
        {
            if (count[s[i] - 'a'] == 1)
                return i;
        }

        return -1;
    }
};
// @lc code=end
