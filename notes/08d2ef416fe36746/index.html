236.二叉树的最近公共祖先

236.二叉树的最近公共祖先.cpp
/*
 * @lc app=leetcode.cn id=236 lang=cpp
 *
 * [236] 二叉树的最近公共祖先
 */

#include "leetcode.h"

// @lc code=start
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if (root == nullptr)
        {
            return nullptr;
        }
        
        if (root == p || root == q) {
            return root;
        }

        // 在左子树找公共祖先
        TreeNode *left = lowestCommonAncestor(root->left, p, q);
        // 在右子树找公共祖先
        TreeNode *right = lowestCommonAncestor(root->right, p, q);

        // 都找不到，则当前节点为公共组件   
        if (left != nullptr && right != nullptr) {
            return root;
        }

        if (left != nullptr) {
            return left;
        }

        if (right != nullptr) {
            return right;
        }
        return nullptr;
    }
};
// @lc code=end
