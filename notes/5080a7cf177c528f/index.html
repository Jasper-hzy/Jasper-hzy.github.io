695.岛屿的最大面积

695.岛屿的最大面积.cpp
/*
 * @lc app=leetcode.cn id=695 lang=cpp
 *
 * [695] 岛屿的最大面积
 */
#include "leetcode.h"
// @lc code=start
class Solution
{
public:
    vector<vector<int>> gridMap;
    int maxAreaOfIsland(vector<vector<int>> &grid, int i, int j)
    {
        if (i < 0 || i >= grid.size() || j < 0 || j >= grid[0].size())
        {
            return 0;
        }
        if (gridMap[i][j] == 1 || grid[i][j] == 0)
        {
            return 0;
        }
        gridMap[i][j] = 1;
        int top = maxAreaOfIsland(grid, i - 1, j);
        int down = maxAreaOfIsland(grid, i + 1, j);
        int left = maxAreaOfIsland(grid, i, j - 1);
        int right = maxAreaOfIsland(grid, i, j + 1);
        int sum = top + down + left + right + 1;
        return sum;
    }

    int maxAreaOfIsland(vector<vector<int>> &grid)
    {
        int maxSize = 0;
        gridMap.assign(grid.size(), vector<int>(grid[0].size(), 0));
        for (int i = 0; i < grid.size(); i++)
        {
            for (int j = 0; j < grid[i].size(); j++)
            {
                int size = maxAreaOfIsland(grid, i, j);
                maxSize = maxSize > size ? maxSize : size;
            }
        }
        return maxSize;
    }
};
// @lc code=end
