3.无重复字符的最长子串

3.无重复字符的最长子串.cpp
/*
 * @lc app=leetcode.cn id=3 lang=cpp
 *
 * [3] 无重复字符的最长子串
 */
#include "leetcode.h"
// @lc code=start
class Solution
{
public:
    int lengthOfLongestSubstring(string s)
    {
        unordered_map<char, int> map;
        int l = 0;
        int r = 0;
        int rs = 0;
        while (r < s.size()) 
        {
            char ch = s[r];
            map[ch] ++;
            while(map[ch] > 1 && l < r) 
            {
                map[s[l]] --;
                l++;
            }
            rs = max(rs, r - l + 1);
            r ++;
        }
        return rs;
    }
};
// @lc code=end
