/*
 * @lc app=leetcode.cn id=739 lang=cpp
 *
 * [739] 每日温度
 */

#include "leetcode.h"

// @lc code=start
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        vector<int> maxTemps;
        vector<int> ans(temperatures.size(), 0);
        for (int i = temperatures.size() - 1; i >= 0; i--)
        {
            while(!maxTemps.empty() && temperatures[maxTemps.back()] <= temperatures[i]) 
            {
                maxTemps.pop_back();
            }
            if (!maxTemps.empty())
            {
                ans[i] = maxTemps.back() - i;
            }
            maxTemps.push_back(i);
       
        }
        return ans;
    }
};
// @lc code=end
