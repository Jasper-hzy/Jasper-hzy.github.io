93.复原 IP 地址

93.复原 IP 地址.cpp
/*
 * @lc app=leetcode.cn id=93 lang=cpp
 *
 * [93] 复原 IP 地址
 */
#include "leetcode.h";
// @lc code=start
class Solution {
public: 

    // 回溯 + 剪枝
    vector<string> rs;
    vector<string> restoreIpAddresses(string s) 
    {
        vector<string> path;
        dbs(s, 0, path);    
        return rs;
    }

    /**
    * s字符串，从start开始，对应拼出的path IP
    */
    void dbs(string s, int start, vector<string>& path) 
    {
        if (path.size() == 4)
        {
            if(start == s.length()) 
            {
                rs.push_back(
                    path[0] + "." + path[1] + "." + path[2] + "." + path[3]
                );
            }
            return;
        }
        if (start > s.length())
        {
            return;
        }
        for(int i = 1; i <= 3 && start + i <= s.length(); i ++)
        {
            int index = start + i;
            string child = s.substr(start, i);
            if((i > 1 && s[start] == '0') || stoi(child) > 255)
            {
                break;
            }
            path.push_back(child);
            dbs(s, index, path);
            path.pop_back();
        }
    }
};
// @lc code=end
