/*
 * @lc app=leetcode.cn id=1 lang=cpp
 *
 * [1] 两数之和
 */
// @lc code=start
#include <vector>
#include <unordered_map>

using namespace std;
class Solution
{
public:
    std::vector<int> twoSum(std::vector<int> &nums, int target)
    {
        unordered_map<int, int> map;
        for (int i = 0; i < nums.size(); i++)
        {
            int cur = nums[i];
            int diff = target - cur;
            if (map.count(diff))
            {
                return {map[diff], i};
            }
            map[cur] = i;
        }
        return {};
    }
};
// @lc code=end
