688.骑士在棋盘上的概率.cpp
/*
 * @lc app=leetcode.cn id=688 lang=cpp
 *
 * [688] 骑士在棋盘上的概率
 */
#include "leetcode.h"
// @lc code=start
class Solution {
public:
    map<tuple<int,int,int>, double> memo;
    double knightProbability(int n, int k, int row, int column) {
        if(row < 0 || row >= n || column < 0 || column >= n) {
            return 0;
        }
        if(k == 0) {
            return 1;
        }
        auto key = make_tuple(k, row, column);
        if (memo.count(key))
            return memo[key];

        double result = 0;
        result += knightProbability(n, k - 1, row - 2, column - 1) * 1 / 8;
        result += knightProbability(n, k - 1, row - 2, column + 1) * 1 / 8;
        result += knightProbability(n, k - 1, row - 1, column - 2) * 1 / 8;
        result += knightProbability(n, k - 1, row - 1, column + 2) * 1 / 8;
        result += knightProbability(n, k - 1, row + 1, column - 2) * 1 / 8;
        result += knightProbability(n, k - 1, row + 2, column - 1) * 1 / 8;
        result += knightProbability(n, k - 1, row + 2, column + 1) * 1 / 8;
        result += knightProbability(n, k - 1, row + 1, column + 2) * 1 / 8;
        return  memo[key] = result;
    }  
};
// @lc code=end
