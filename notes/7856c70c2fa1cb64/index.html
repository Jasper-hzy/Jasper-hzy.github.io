1、 新建gradle插件工程

引入依赖

groovy
plugins {
    id 'java'
    id 'java-gradle-plugin'
    id 'org.jetbrains.kotlin.jvm'
    id 'kotlin-kapt'
}

compileKotlin {
    kotlinOptions.jvmTarget = "1.8"
}

dependencies {
    implementation "org.jetbrains.kotlin:kotlin-stdlib:$kotlin_version"
    implementation "org.jetbrains.kotlin:kotlin-gradle-plugin-api:$kotlin_version"
    implementation "org.jetbrains.kotlin:kotlin-gradle-plugin:$kotlin_version"
    compileOnly 'com.android.tools.build:gradle:3.5.3'
    // 用来自动化化注册处理器的工具
    compileOnly "com.google.auto.service:auto-service:1.0-rc7"
    kapt "com.google.auto.service:auto-service:1.0-rc7"
}

// 设置gradle plugin插件id & 插件对应实现类
gradlePlugin {
    plugins {
        simplePlugin {
            id = 'kcp-plugin'
            implementationClass = 'com.example.kcp.KcpGradleSupportPlugin'
        }
    }
}

插件实现类实现

  • 第一种可以直接使用KotlinCompilerPluginSupportPlugin的子类实现，然后指定为插件的实现类，如上面这个com.example.kcp.KcpGradleSupportPlugin类。

    kotlin
    class KcpGradleSupportPlugin : KotlinCompilerPluginSupportPlugin {

        override fun apply(target: Project) {
            super.apply(target)
            // 创建extension
            target.extensions.create(
                    KcpExtension::class.java,
                    "kcp",
                    KcpExtensionImpl::class.java,
                    object : Action<KcpExtensionImpl> {
                        override fun execute(extImpl: KcpExtensionImpl) {
                            println("forTest-测试触发KcpExtensionImpl  ${extImpl.enabled} ${extImpl.kcpConfig.version}")
                        }
                    }
            )
        }

        override fun applyToCompilation(kotlinCompilation: KotlinCompilation<*>): Provider<List<SubpluginOption>> {
            // 读取插件 extension 传递 extension作为 编译插件 参数值
            val project = kotlinCompilation.target.project
            return project.provider { emptyList()
    //            mutableListOf(SubpluginOption(key = "enabled", value = "true"))
            }
        }

        override fun getCompilerPluginId(): String {
            return PLUGIN_ID
        }

override fun getPluginArtifact(): SubpluginArtifact { return SubpluginArtifact( groupId = GROUP_ID, artifactId = PLUGIN_ID, version = VERSION ) }

override fun getPluginArtifactForNative(): SubpluginArtifact = SubpluginArtifact( groupId = GROUP_ID, artifactId = PLUGIN_ID, version = VERSION )

override fun isApplicable(kotlinCompilation: KotlinCompilation<*>): Boolean = true }

code
  - 其中有几个注意的点：
    - getCompilerPluginId()   这个方法需要返回一个插件的ID，getPluginArtifact() 方法会放返回插件引用路径，在编译时会自动从本地仓库路径去引入GROUP_ID:PLUGIN_ID:VERSION这个库作为classpath，从而引入我们自己的构建插件（这个时候工程还没有构建）
      - 当我们需要使用gradle extension去传递构建参数时，可以在applyToCompilation这个方法去传递Option，具体解析格式在创建编译插件工程时会涉及到。

- 第二种是使用传统的继承Plugin类，SubPlugin使用AutoService注册

  ```kotlin
  class KcpGradlePlugin : Plugin<Project> {
  
      override fun apply(project: Project) {
          project.extensions.create(
                  KcpExtension::class.java,
                  "kcp",
                  KcpExtensionImpl::class.java,
                  object : Action<KcpExtensionImpl> {
                      override fun execute(extImpl: KcpExtensionImpl) {
                      }
                  }
          )
      }
  }
groovy
  gradlePlugin {
      plugins {
          simplePlugin {
              id = 'kcp-plugin'
              // 这个的实现类路径替换成上面这个类的路径
              implementationClass = 'com.example.kcp.KcpGradlePlugin'
          }
      }
  }

原来的SubPlugin类需要添加一个@AutoService注解

kotlin
  @AutoService(KotlinCompilerPluginSupportPlugin::class)
  class KcpGradleSupportPlugin : KotlinCompilerPluginSupportPlugin()

plugins {

    kotlinOptions.jvmTarget = "1.8"
}
compileTestKotlin {
    kotlinOptions.jvmTarget = "1.8"
}

dependencies {
    implementation "org.jetbrains.kotlin:kotlin-stdlib:$kotlin_version"
    // kcp依赖
    compileOnly "org.jetbrains.kotlin:kotlin-compiler-embeddable:$kotlin_version"

    // 用来自动化化注册处理器的工具
    compileOnly "com.google.auto.service:auto-service:1.0-rc7"
    kapt "com.google.auto.service:auto-service:1.0-rc7"

    testImplementation "junit:junit:4.13.2"
    // 用来自动化测试kcp生成代码
    testImplementation("com.github.tschuchortdev:kotlin-compile-testing:1.4.2")
    testImplementation "com.google.truth:truth:1.1.2"
}

实现ComponentRegistrar子类

该子类用于注册编译阶段中的自定义步骤拦截

kotlin
@AutoService(ComponentRegistrar::class)
class MComponentRegistrar : ComponentRegistrar {
    override fun registerProjectComponents(project: MockProject, configuration: CompilerConfiguration) {
        val messageCollector = requireNotNull(configuration[CLIConfigurationKeys.MESSAGE_COLLECTOR_KEY]) {
            "failed to find message collector"
        }
        // 注册编译步骤
        // 1. 注册语法解析流程、 此流程在编译前端 生成语法树之后、可以用来收集类信息做前置校验
        AnalysisHandlerExtension.registerExtension(
                project,
                MAnalysisHandlerExtension(messageCollector) // 语法分析
        )
        ClassBuilderInterceptorExtension.registerExtension(
                project,
                MClassBuilderInterceptorExtension(messageCollector) // 类信息构建
        )
        ExpressionCodegenExtension.registerExtension(
                project,
                MExpressionCodegenExtension(messageCollector) // 类生成代码
        )

        // IR编译适配 todo
    }
}

buildscript {

    id 'java'
    id 'org.jetbrains.kotlin.jvm'
}

// 替换为本地依赖
// 由于SubPlugin会自动从仓库下载我们之前传递的插件库，这里替换指向为本地工程
configurations.all {
    resolutionStrategy.dependencySubstitution {
        substitute module("com.example.tools.kcp:kcp-plugin") using project(":AopDemo:kcp:kotlin-compiler-plugin") because "using local version"
    }
}


gradle.projectsEvaluated {
    tasks.findByName("build")
            .dependsOn(project(":AopDemo:kcp:kotlin-compiler-plugin").tasks.findByName("build"))
            .dependsOn(project(":AopDemo:kcp:kcp-gradle-plugin").tasks.findByName("build"))
}

//需要 build kcp-gradle-plugin 这个仓库否则这里会报错

kcp {
    enabled = true
    config {
        version = 1234
    }
