/*
 * @lc app=leetcode.cn id=146 lang=cpp
 *
 * [146] LRU 缓存
 */
#include "leetcode.h"
// @lc code=start
class LRUCache
{
public:
    size_t capacity_;
    list<pair<int, int>> nodes_;
    unordered_map<int, list<pair<int, int>>::iterator> key_to_nodes_;

    LRUCache(int capacity) : capacity_(capacity)
    {
    }

    int get(int key)
    {
        auto cache_value = key_to_nodes_.find(key);
        if (cache_value != key_to_nodes_.end())
        {
            nodes_.splice(nodes_.begin(), nodes_, cache_value->second);
            return cache_value->second->second;
        }
        return -1;
    }

    void put(int key, int value)
    {
        auto cache_value = key_to_nodes_.find(key);
        if (cache_value != key_to_nodes_.end())
        {
            cache_value->second->second = value;
            nodes_.splice(nodes_.begin(), nodes_, cache_value->second);
            return;
        }
        if (key_to_nodes_.size() >= capacity_)
        {
            int remove_key = nodes_.back().first;
            nodes_.pop_back();
            key_to_nodes_.erase(remove_key);
        }
        nodes_.push_front({key, value});
        key_to_nodes_[key] = nodes_.begin();
    }
};
// @lc code=end
