滑动窗口

保留题目对应的 C++ 练习实现，便于回顾与继续练习。

滑动窗口.cpp
#include <algorithm>
#include <climits>
#include <iostream>
#include <string>
#include <vector>

using namespace std;

class SlidingWindowTemplate {
public:
    // 固定长度窗口：438. 找到字符串中所有字母异位词
    vector<int> findAnagrams(string s, string p) {
        vector<int> ans;

        if (s.size() < p.size()) {
            return ans;
        }

        vector<int> need(26, 0);
        vector<int> window(26, 0);

        for (char c : p) {
            need[c - 'a']++;
        }

        // left / right 始终表示当前窗口的左右边界。
        // 固定窗口也优先用这套写法，不用额外记 i - len 公式。
        int left = 0;
        int k = static_cast<int>(p.size());

        for (int right = 0; right < static_cast<int>(s.size()); ++right) {
            window[s[right] - 'a']++;

            while (right - left + 1 > k) {
                window[s[left] - 'a']--;
                ++left;
            }

            if (right - left + 1 == k && window == need) {
                ans.push_back(left);
            }
        }

        return ans;
    }

    // 可变长度窗口，求最长：3. 无重复字符的最长子串
    int lengthOfLongestSubstring(string s) {
        vector<int> count(128, 0);
        int left = 0;
        int ans = 0;

        for (int right = 0; right < static_cast<int>(s.size()); ++right) {
            count[s[right]]++;

            while (count[s[right]] > 1) {
                count[s[left]]--;
                ++left;
            }

            ans = max(ans, right - left + 1);
        }

        return ans;
    }

    // 可变长度窗口，求最短：209. 长度最小的子数组
    int minSubArrayLen(int target, vector<int>& nums) {
        int left = 0;
        int sum = 0;
        int ans = INT_MAX;

        for (int right = 0; right < static_cast<int>(nums.size()); ++right) {
            sum += nums[right];

            while (sum >= target) {
                ans = min(ans, right - left + 1);
                sum -= nums[left];
                ++left;
            }
        }

        return ans == INT_MAX ? 0 : ans;
    }
};

static void printVector(const vector<int>& nums) {
    cout << "[";
    for (int i = 0; i < static_cast<int>(nums.size()); ++i) {
        if (i > 0) {
            cout << ", ";
        }
        cout << nums[i];
    }
    cout << "]";
}

int main() {
    SlidingWindowTemplate solution;

    vector<int> anagrams = solution.findAnagrams("cbaebabacd", "abc");
    cout << "438 输出: ";
    printVector(anagrams);
    cout << "，期望: [0, 6]" << endl;

    cout << "3 输出: " << solution.lengthOfLongestSubstring("abcabcbb")
         << "，期望: 3" << endl;

    vector<int> nums = {2, 3, 1, 2, 4, 3};
    cout << "209 输出: " << solution.minSubArrayLen(7, nums)
         << "，期望: 2" << endl;

    return 0;
}
