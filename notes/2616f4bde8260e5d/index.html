121.买卖股票的最佳时机

121.买卖股票的最佳时机.cpp
/*
 * @lc app=leetcode.cn id=121 lang=cpp
 *
 * [121] 买卖股票的最佳时机
 */

#include "leetcode.h"

// @lc code=start
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        if(prices.empty())
        {
            return 0;
        }
        int min_price = prices[0];
        int ans = 0;
        for (int i = 1; i < prices.size(); i++)
        {
            ans = max(ans, prices[i] - min_price);
            min_price = min(min_price, prices[i]);
        }
        return  ans;

    }
};
// @lc code=end
