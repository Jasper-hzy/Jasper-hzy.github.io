三线程顺序打印 ABC：Java 练习代码

通过三个信号量传递打印许可，形成 A → B → C 的顺序。

ABCPrinterPractice.java
import java.util.concurrent.Semaphore;

/**
 * 三线程顺序打印 ABC。
 *
 * 目标输出：ABCABCABCABCABC
 */
public class ABCPrinterPractice {

    private static final int LOOP_COUNT = 5;

    private static final Semaphore semaphoreA = new Semaphore(1);
    private static final Semaphore semaphoreB = new Semaphore(0);
    private static final Semaphore semaphoreC = new Semaphore(0);

    private static void print(char ch, Semaphore cur, Semaphore next) throws InterruptedException {
        for (int i = 0; i < LOOP_COUNT; i++) {
            cur.acquire();
            System.out.print(ch);
            next.release();
        }
    }

    public static void main(String[] args) throws InterruptedException {
        Thread threadA = new Thread(() -> {
            try {
                print('A', semaphoreA, semaphoreB);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }, "A");

        Thread threadB = new Thread(() -> {
            try {
                print('B', semaphoreB, semaphoreC);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }, "B");

        Thread threadC = new Thread(() -> {
            try {
                print('C', semaphoreC, semaphoreA);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }, "C");

        threadA.start();
        threadB.start();
        threadC.start();

        threadA.join();
        threadB.join();
        threadC.join();

        System.out.println();
        System.out.println("done");
    }
}
