25.K-个一组翻转链表.cpp
/*
 * @lc app=leetcode.cn id=25 lang=cpp
 *
 * [25] K 个一组翻转链表
 */

#include "leetcode.h"

// @lc code=start
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:

    // 反转 [a, b)
    ListNode* reverseList(ListNode* start, ListNode* end) {
        if(start == nullptr)
        {
            return start;
        }
        ListNode* cur = start;
        ListNode* pre = nullptr;
        while(cur != end)
        {
            ListNode* next_node = cur->next;
            cur->next = pre;
            pre = cur;
            cur = next_node;
        }
        return pre;
    }

    ListNode* reverseKGroup(ListNode* head, int k) {
        if(head == nullptr)
        {
            return head;
        }

        ListNode* start = head;
        ListNode* next_start = head;
        for (int i = k; i > 0; --i)
        {
            if(next_start == nullptr)
            {
                return start;
            }
            next_start = next_start->next;
        }
        ListNode* new_start = reverseList(start, next_start);
        start->next = reverseKGroup(next_start, k);
        return new_start;
    }
};
// @lc code=end
